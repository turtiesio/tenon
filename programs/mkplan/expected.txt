# nothing exists yet: everything is built, dependencies first
$ mkplan fresh.mk
[main.o] cc -c main.c
[util.o] cc -c util.c
[app] cc -o app main.o util.o
3 to build, 0 up to date
[exit 0]
# all up to date
$ mkplan uptodate.mk
0 to build, 3 up to date
[exit 0]
# one source is newer: rebuilds propagate transitively
$ mkplan newer.mk
[util.o] cc -c util.c
[app] cc -o app main.o util.o
2 to build, 1 up to date
[exit 0]
# equal mtime is not newer
$ mkplan equal.mk
0 to build, 1 up to date
[exit 0]
# a target with no mtime is rebuilt even if its deps are old
$ mkplan nomtime.mk
[out] make out
1 to build, 0 up to date
[exit 0]
# TARGET argument selects a sub-plan; the rest is ignored
$ mkplan newer.mk util.o
[util.o] cc -c util.c
1 to build, 0 up to date
[exit 0]
$ mkplan newer.mk main.o
0 to build, 1 up to date
[exit 0]
$ mkplan newer.mk app
[util.o] cc -c util.c
[app] cc -o app main.o util.o
2 to build, 1 up to date
[exit 0]
$ mkplan newer.mk main.c
mkplan: unknown target main.c
[exit 1]
$ mkplan newer.mk nothing
mkplan: unknown target nothing
[exit 1]
# ties broken by order of appearance, not by dependency order

[a] echo a
[b] echo b
[all] echo all

[exit 0]
# diamond: the shared dependency is planned once
$ mkplan diamond.mk

[left] echo left
[right] echo right
[top] echo top
4 to build, 0 up to date
[exit 0]

[exit 0]
